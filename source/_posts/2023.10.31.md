def distinct(id_batch, id_query, rule_list):
    try:
        # 尝试获取远程服务器的代理，从而跟远程服务器进行通信
        server_proxy = RpcHandler.get_server_proxy()
        task_name = 'flow_distinct_all'
        for rule_id in rule_list:
            task_id = '%s_%s' % (rule_id, id_batch)
            data_path = os.path.join(FSystem.BACKUP_AREA, id_batch, id_query, rule_id)
            task_args = {'data_path': data_path, 'query_id': id_query, 'rule_id': rule_id,
                         'batch_id': id_batch, 'user_name': 'admin'}
            server_proxy.task_add(task_id)
            server_proxy.executor(task_name, task_id, task_args)
            logger.info("Task: %s Submit." % task_id)
    except Exception as ex:
        logger.exception(ex)

def executor(id_batch, id_query, rule_list):
    try:
        server_proxy = RpcHandler.get_server_proxy()
        for rule_id in rule_list:
            executor_model = DBHelper().query_for_operation(rule_id, id_batch, id_query)
            if executor_model is None:
                continue
            DBHelper().update_state_for_rule(executor_model.rule_summary_id, RuleState.DEALING)
            exec_handler = ExecutorHandler(executor_model.rule_summary_id,executor_model.id_query,executor_model.id_rule,
                                           executor_model.id_batch,executor_model.user_name,executor_model.user_id)
            operations = executor_model.operations
            if Pretreatment.BUSINESS_ANALYSIS in operations.keys():
                operation_id = operations.get(Pretreatment.BUSINESS_ANALYSIS)[0]
                operation_attach = operations.get(Pretreatment.BUSINESS_ANALYSIS)[1]
                exec_handler.executor_for_business(operation_id, ip_location=bool(int(operation_attach)))
            recover_state = False
            if Pretreatment.DATA_RECOVER in operations.keys():
                operation_id = operations.get(Pretreatment.DATA_RECOVER)[0]
                recover_state = exec_handler.executor_for_recover(is_recover=True, id_operation=operation_id)
            elif Pretreatment.PROTOCOL_PARSER in operations.keys() or Pretreatment.LINKAGE in operations.keys():
                recover_state = exec_handler.executor_for_recover(is_recover=False)
            if recover_state is False:
                logger.info("Need not Recover Or Recover Failure.")
            else:
                if Pretreatment.PROTOCOL_PARSER in operations.keys():
                    exec_handler.executor_for_protocol(operations.get(Pretreatment.PROTOCOL_PARSER)[0])
                if Pretreatment.LINKAGE in operations.keys():
                    exec_handler.executor_for_linkage(operations.get(Pretreatment.LINKAGE)[0])
            while True:
                task_id = "%s_%s" % (executor_model.id_rule, executor_model.id_batch)
                print(task_id)
                counter = server_proxy.task_status(task_id)
                print(counter)
                if counter <= 0:
                    break
                time.sleep(10)
            DBHelper().update_state_for_rule(executor_model.rule_summary_id, RuleState.FINISH)
        FSHandler.delete_file(os.path.join(WorkSpace.HANDLE_AREA, id_batch))
        FSHandler.delete_file("%s/*" % WorkSpace.RESULT_AREA)
    except Exception as ex:
        logger.exception(ex)

def analysis():
    ready_tasks = list()
    try:
        handler = RedisHandler()
        for key in handler.query_keys():
            expected_host = handler.query_by_key(key).get('host')
            expected_state = handler.query_by_key(key).get('state')
            if expected_host != consumer_host or expected_state != TaskState.READY:
                continue
            ready_tasks.append(handler.query_by_key(key))
        for ready_task in ready_tasks:
            id_query,file_path,is_zip = ready_task.get('id_query'),ready_task.get('file_path'),ready_task.get('is_zip')
            data_time = ready_task.get('data_time')
            HDFSHandler.download_file(file_path, WorkSpace.HANDLE_AREA)
            file_handler = FileHandler(id_query, os.path.split(file_path)[1], WorkSpace.HANDLE_AREA)
            decompress_state = file_handler.decompress(is_zip)
            if decompress_state is False:
                logger.info("File %s Decompress Failure." % file_handler.file_name)
                FSHandler.delete_file(os.path.join(file_handler.file_path, file_handler.file_name))

class TimeUtil:
    __slots__ = ['year', 'month', 'day', 'hour', 'minute', 'second', 'time_format']

    def __init__(self):
        self.year = ''
        self.month = ''
        self.day = ''
        self.hour = ''
        self.minute = ''
        self.second = ''
        self.time_format = '%Y-%m-%d %H:%M:%S'

    def __convert_datetime(self, time_str):
        self.year = time_str[:4]
        self.month = time_str[4:6]
        self.day = time_str[6:8]
        self.hour = time_str[8:10]
        self.minute = time_str[10:12]
        self.second = time_str[12:14]
        covert_time = '%s-%s-%s %s:%s:%s' % (self.year, self.month, self.day, self.hour, self.minute, self.second)
        return datetime.strptime(covert_time, self.time_format)

    def get_current_time(self):
        current_time = datetime.now()
        current_time.strftime(self.time_format)
        return current_time

    def get_time_interval(self, str_time):
        return (self.get_current_time() - self.__convert_datetime(str_time)).days

    def get_update_time(self, time_str, time_scope):
        current_time = self.__convert_datetime(time_str)
        next_time_repr = str(current_time + timedelta(days=time_scope))
        next_time_str = next_time_repr.replace('-', '').replace(' ', '').replace(':', '')
        return next_time_repr, next_time_str

    def query_keys(self):
        """
        :return: 返回值是一个列表
        """
        # todo: self.conn.keys() 方法会返回 Redis 数据库中所有的键名，以列表形式返回。
        # todo: decode() 方法可以将一个字节串解码为字符串，这里的作用是将 Redis 数据库中保存的二进制字符串转换为 Unicode 字符串。
        # todo: map函数会对后面的列表中的每个键都调用Lambda方法，将每个键都进行解码，并将解码后的字符串还放到一个列表里面list(1,2,3,4)
        return list(map(lambda x: x.decode(), self.conn.keys()))

    def query_by_key(self, key):
        # todo:
        return eval(self.conn.get(key.encode()).decode())

    def update_by_key(self, key, item_dict):
        current_value = self.query_by_key(key)
        for item_key in item_dict:
            current_value[item_key] = item_dict[item_key]
        self.insert(key, current_value)

    def delete_by_key(self, key):
        self.conn.delete(key.encode())

            target_file = os.path.join(self.file_path, self.file_name)

                if is_zip:
                    os.system("unzip %s -d %s" % (target_file, self.file_path))
                    os.system("rm -rf %s" % target_file)
                decompress_state = True
        except Exception as ex:
            logger.exception(ex)
        finally:
            return decompress_state

    def extract_by_rule(self, is_zip):
        extract_state = False
        file_name = os.path.splitext(self.file_name)[0] if is_zip else self.file_name
        target_path = os.path.join(self.file_path, file_name)
        try:
            process_list = list()
            with open(target_path, encoding='UTF-8') as reader:
                record = reader.readline()
                while True:
                    if not record:
                        break
                    rule_id = record.split('\t')[1]
                    if rule_id != '事件名' and rule_id not in self.rule_list:
                        self.rule_list.append(rule_id)
                        process = Process(target=self.__generate_by_rule, args=(rule_id, target_path))
                        process_list.append(process)
                        process.start()
                    record = reader.readline()
            for process in process_list:
                process.join()
            extract_state = True
        except Exception as ex:
            logger.exception(ex)
        finally:
            FSHandler.delete_file(target_path)
            return extract_state

    def __generate_by_rule(self, rule_id, src_file):
        rule_path = os.path.join(self.file_path, self.id_batch, self.id_query, rule_id)
        if not os.path.exists(rule_path):
            FSHandler.make_dir(rule_path)
        rule_records = list()
        with open(src_file, encoding='UTF-8') as reader:
            recorder = reader.readline()
            while True:
                if not recorder:
                    break
                if rule_id == recorder.split('\t')[1]:
                    if len(rule_records) <= self.file_unit:
                        rule_records.append(recorder)
                    else:
                        self.__data_persist(rule_records, rule_path, rule_id)
                        rule_records.clear()
                recorder = reader.readline()
        if len(rule_records) > 0:
            self.__data_persist(rule_records, rule_path, rule_id)
            rule_records.clear()
        logger.info("Rule %s File Generate Success." % rule_id)

    def __data_persist(self, data_list, data_path, data_rule):
        time_str = time.strftime("%H%M%S", time.localtime(time.time()))
        random_str = str(random.randint(1, 99999))
        data_file_name = '%s_%s_%s_%s.txt' % (data_rule, self.id_query, time_str, random_str)
        data_file_path = os.path.join(data_path, data_file_name)
        with open(data_file_path, 'w+', encoding='UTF-8') as writer:
            for data_recorder in data_list:
                writer.write(data_recorder)
        logger.info("%s Generate Success." % data_file_name)

    def __init__(self, rule_summary_id, operation_id, id_query, id_rule, id_batch, user_name, user_id):

        try:
            self.ftp_client = ftplib.FTP(FTPConfig.HOST)
            self.ftp_client.login()
        except Exception as ex:
            logger.exception(ex)

    def __close(self):
        if self.ftp_client is not None:
            self.ftp_client.close()

    def __compress(self):
        local_target_file = None
        try:
            local_file_path = os.path.join(WorkSpace.RESULT_AREA, self.user_name, self.id_query, self.id_rule,
                                           self.id_batch)
            local_target_filename = '%s_%s_%s_%s_%s_%s_%s.zip' % (self.user_name, self.user_id, self.rule_summary_id,
                                                                  self.operation_id, self.id_query, self.id_rule,
                                                                  self.id_batch)
            local_target_file = os.path.join(local_file_path, local_target_filename)
            os.system("zip -j %s %s/*" % (local_target_file, local_file_path))
        except Exception as ex:
            local_target_file = None
            logger.exception(ex)
        finally:
            return local_target_file

    def upload(self):
        upload_state = False
        try:
            local_target_file = self.__compress()
            self.__login()
            if local_target_file is not None:
                buf_size = 1024
                file_cursor = open(local_target_file, 'rb')
                self.ftp_client.storbinary('STOR ' + os.path.split(local_target_file)[1], file_cursor, buf_size)
                self.__close()
                os.system("rm -rf %s" % local_target_file)
                upload_state = True
        except Exception as ex:
            logger.exception(ex)
        finally:
            return upload_state

    def secure_copy(self, remote_host, remote_path):
        upload_state = False
        try:
            local_target_file = self.__compress()
            if local_target_file is not None:
                FSHandler.upload_file(local_target_file, remote_host, remote_path)
                FSHandler.delete_file(local_target_file)
                upload_state = True
        except Exception as ex:
            logger.exception(ex)
        finally:
            return upload_state
