class Logging:
    """日志处理类"""
    def __init__(self):
        self.logger = logging.getLogger(__name__)

    def __config(self):
        """配置相关属性"""
        formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s')
        console_handler = logging.StreamHandler(sys.stdout)
        console_handler.formatter = formatter
        file_handler = logging.FileHandler('SparkRPC8093.log',encoding='UTF-8')
        file_handler.setFormatter(formatter)
        self.logger.addHandler(console_handler)
        self.logger.addHandler(file_handler)
        self.logger.setLevel(logging.INFO)

    def get_logger(self):
        self.__config()
        return self.logger


global_logger = Logging().get_logger()


def time_format(datetime_stamp):
    year, month, day = datetime_stamp[:4], datetime_stamp[4:6], datetime_stamp[6:8]
    hour, minute, second = datetime_stamp[8:10], datetime_stamp[10:12], datetime_stamp[12:14]
    return '%s-%s-%s %s:%s:%s' % (year, month, day, hour, minute, second)

class DBHandler:
    def __init__(self, user_name):
        self.conn = None
        self.user_name = user_name

    def __connect(self):
        self.conn = pymysql.connect(host=DBConfig.HOST, user=DBConfig.USER, password=DBConfig.PASS,
                                    database=self.user_name, charset=DBConfig.CHARSET)

    def __close(self):
        if self.conn is not None:
            self.conn.close()

    def __enter__(self):
        self.__connect()
        return self.conn

    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.conn is not None:
            self.conn.close()
        self.conn = None

class Scheduler:

    def __init__(self):
        self.task_id = 'task_id'
        self.task_name = 'task_name'
        self.task_args = 'task_args'
        self.task_list = list()
        self.task_counter = dict()

    def receiver(self, task_name, task_id, task_args):
        global_logger.info("Receive Task: %s" % task_id)
        task = {self.task_name: task_name, self.task_id: task_id, self.task_args: task_args}
        self.task_list.append(task)
        return True

    def executor(self):
        while True:
            if len(self.task_list) is 0:
                time.sleep(5)
                continue
            task = self.task_list.pop(0)
            global_logger.info(task)
            task_id = task.get(self.task_id)
            task_args = task.get(self.task_args)
            data_path = task_args.get('data_path') if task_args.get('data_path') is not None else ''
            rule_summary_id = task_args.get('rule_summary_id') if task_args.get('rule_summary_id') is not None else ''
            operation_id = task_args.get('operation_id') if task_args.get('operation_id') is not None else ''
            user_id = task_args.get('user_id') if task_args.get('user_id') is not None else ''
            query_id = task_args.get('query_id') if task_args.get('query_id') is not None else ''
            rule_id = task_args.get('rule_id') if task_args.get('rule_id') is not None else ''
            batch_id = task_args.get('batch_id') if task_args.get('batch_id') is not None else ''
            user_name = task_args.get('user_name') if task_args.get('user_name') is not None else ''
            is_location = task_args.get('location') if task_args.get('location') is not None else ''
            data_store_detail = ''
            if task.get(self.task_name) == AppConfig.FLOW_DISTINCT_ALL:
                spark_cmd = basic_run_cmd(AppConfig.FLOW_DISTINCT_ALL)
                spark_cmd += ' %s %s %s %s %s' % (data_path, query_id, rule_id, batch_id, user_name)
                os.system(spark_cmd)
            elif task.get(self.task_name) == AppConfig.FLOW_LOCATION:
                spark_cmd = basic_run_cmd(AppConfig.FLOW_LOCATION)
                spark_cmd += ' %s %s %s %s %s %s' % (data_path, query_id, rule_id, batch_id, user_name, is_location)
                os.system(spark_cmd)
            elif task.get(self.task_name) == AppConfig.FLOW_PROTOCOL:
                spark_cmd = basic_run_cmd(AppConfig.FLOW_PROTOCOL)
                spark_cmd += ' %s %s %s %s %s' % (data_path, query_id, rule_id, batch_id, user_name)
                os.system(spark_cmd)
            elif task.get(self.task_name) == AppConfig.FLOW_LINKAGE:
                spark_cmd = basic_run_cmd(AppConfig.FLOW_LINKAGE)
                spark_cmd += ' %s %s %s %s %s' % (data_path, query_id, rule_id, batch_id, user_name)
                os.system(spark_cmd)
            if rule_summary_id != '' and operation_id != '' and user_id != '':
                if task.get(self.task_name) == AppConfig.FLOW_LOCATION:
                    data_store_detail = 'flow_detail'
                elif task.get(self.task_name) == AppConfig.FLOW_PROTOCOL:
                    data_store_detail = 'protocol_original'
                elif task.get(self.task_name) == AppConfig.FLOW_LINKAGE:
                    data_store_detail = 'flow_security'
                if data_store_detail != '':
                    DBHelper.rule_operation_update(rule_summary_id, operation_id, user_id,
                                                   data_store_detail=data_store_detail)
            self.task_counter[task_id] -= 1
            time.sleep(5)

    def add_task(self, task_id):
        if self.task_counter.get(task_id) is None:
            self.task_counter[task_id] = 0
        self.task_counter[task_id] += 1
        return True

    def del_task(self, task_id):
        self.task_counter[task_id] -= 1
        return False

    def task_state(self, task_id):
        return self.task_counter.get(task_id)


if __name__ == '__main__':
    scheduler = Scheduler()
    thread = Thread(target=scheduler.executor)
    thread.start()
    server = ThreadXMLRPCServer((RpcConfig.RPC_HOST, RpcConfig.RPC_PORT), allow_none=True)
    server.register_function(scheduler.add_task, 'task_add')
    server.register_function(scheduler.del_task, 'task_del')
    server.register_function(scheduler.receiver, "executor")
    server.register_function(scheduler.task_state, 'task_status')
    server.serve_forever()
